# M4, discovery: ReactOS's smss.exe starts under the system DLL with the
# parameters of the first process; the kernel logs the system services it
# calls and how it stops (docs/compat/m4-smss.md). It waits for csrss and
# winlogon, which it cannot start yet: the kernel ends it after 10 s.
# Skipped until the userland is fetched.
features = test-smss
requires = userland
cpus     = default, intel
timeout  = 180

expect = [m4] SMSS.EXE under the system DLL: exit status
# Where it stops today: it reads Session Manager from the SYSTEM hive (Cm),
# creates its object directories and DOS device links (Ob), then cannot open
# its KnownDlls directory on disk (no Io yet). Update with each step of M4.
expect = [boot] SYSTEM hive mounted at \REGISTRY\MACHINE\SYSTEM
expect = [m4]   NtEnumerateValueKey (0x004d)
expect = [m4]   NtCreateSymbolicLinkObject (0x0036)
expect = SMSS: Unable to open a handle to the KnownDll directory
expect = kernel: phase 1 done

# A fatal error ends the run at once.
forbid = *** STOP:
forbid = [m4] SMSS.EXE: the process did not start
forbid = [m4] SMSS.EXE: no image section
forbid = [boot] SYSTEM hive refused
