# Spike S1 (M3): a native program compiled with MinGW from tests/native/ and
# linked to ReactOS's ntdll only. ntdll's LdrInitializeThunk initializes its
# process, binds its imports and enters NtProcessStartup with the PEB, which
# the program prints with DbgPrint before ending with status 0x5152. Skipped
# without the userland (cargo xtask fetch-userland) or without MinGW.
requires = userland, native
cpus     = default, intel
timeout  = 120

expect = [s1] hello from a MinGW native program, PEB 7FFDF000
expect = [s1] S1HELLO.EXE under the system DLL: exit status 0x00005152 (the program ran to its end)
expect = kernel: phase 1 done

# A fatal error ends the run at once.
forbid = *** STOP:
forbid = [s1] S1HELLO.EXE: the process did not
forbid = [s1] S1HELLO.EXE: no image section
