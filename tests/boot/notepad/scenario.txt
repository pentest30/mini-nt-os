# of winlogon.exe, takes WinSta0\Default and starts ReactOS's notepad.exe,
# then lists the top-level windows each second for 8 seconds (the probe of
# smss ends everything 10 seconds after the last key). Notepad shows its
# window by SetWindowPlacement, with the place GetWindowPlacement gave.

cpus = default, intel
timeout = 120

expect = [notepad] on WinSta0\Default

expect = [notepad] probe done
