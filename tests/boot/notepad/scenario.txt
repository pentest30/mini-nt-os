# Notepad, the probe (tests/native/notepad1.c): NOTEPAD1.EXE takes the place
# of winlogon.exe, takes WinSta0\Default, checks RegisterWindowMessage and
# starts ReactOS's notepad.exe, then lists the top-level windows four times a
# second for 8 seconds (the probe of smss ends everything 10 seconds after the
# last key). Notepad shows its window by SetWindowPlacement, with the place
# GetWindowPlacement gave; GetWindowThreadProcessId (NtUserQueryWindow) gives
# its process and thread. Ctrl+G, typed by QEMU, is one of Notepad's
# accelerators ("^G": WM_CHAR 7, TranslateAccelerator): its "Goto line" dialog;
# Escape closes it. Before it, QEMU types two lines into Notepad's Edit
# control, which places and draws them by Uniscribe (usp10: GetFontData,
# GetGlyphIndicesW, GetCharABCWidthsI, ExtTextOut with ETO_GLYPH_INDEX).
features = test-smss
requires = userland, native
initial-command = NOTEPAD1.EXE
cpus = default, intel
timeout = 120

keys = [notepad] Notepad is up => "Hello, Mino!" ret "Line 2" ctrl-g
keys = [notepad] the Goto line dialog is up => esc

expect = [notepad] on WinSta0\Default
expect = [notepad] RegisterWindowMessage: one message per name
expect = [notepad] launched (PID
expect = "Untitled - Notepad"
expect = [notepad] Notepad is up: its window is its main thread's
expect = "Goto line"
expect = [notepad] the Goto line dialog is up
expect = [notepad] the Goto line dialog is closed
expect = [notepad] probe done

forbid = *** STOP:

forbid = sys: NtUserTranslateAccelerator (0x1234) is not implemented
forbid = sys: NtUserQueryWindow (0x11e1) is not implemented
forbid = sys: NtUserRegisterWindowMessage (0x11eb) is not implemented

forbid = sys: NtGdiGetGlyphIndicesW (0x10b2) is not implemented
forbid = sys: NtGdiGetCharABCWidthsW (0x109f) is not implemented
