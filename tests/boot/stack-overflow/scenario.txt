# Kernel stack overflow (M0): the boot thread overflows its guarded kernel
# stack right after switching to it; the boot must end on a readable STOP
# screen naming the overflow, not on a triple fault (QEMU runs -no-reboot).
# Its call stack (M1) starts in the recursion that overflowed.
features = test-stack-overflow
cpus     = default
timeout  = 120

expect = *** STOP: 0x0000007f
expect = KERNEL STACK OVERFLOW: guard page hit
expect = ntoskrnl::recurse+0x
expect = more times)
expect = ntoskrnl::provoke_stack_overflow+0x
expect = System halted.

# The double fault stops, not a panic (KeBugCheckEx 0x1E), and the screen
# itself does not fault.
forbid = KMODE_EXCEPTION_NOT_HANDLED
forbid = STOP while printing the STOP screen
