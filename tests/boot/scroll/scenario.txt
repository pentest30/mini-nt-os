# The scroll bars of windows (win32k::scroll; tests/native/scroll1.c, MinGW,
# linked to ReactOS's kernel32 and user32): SCROLL1.EXE takes the place of
# winlogon.exe and switches to WinSta0\Default. C, without a scroll bar
# style, has no scroll bars; A (100, 100, 400 x 300) has its bars from
# WM_NCCREATE (user32's DefWindowProc: 0 to 100), sets its vertical bar from 0 to 99, a
# page of 10, at 30, which user32 reads back from the SBINFO in the
# desktop's heap, then past the end: it stops at 90; B (600, 100) loses its
# horizontal bar when the bar has nothing to scroll, and so on. A click on
# the down arrow of A's vertical bar (at 488, 373; the pointer starts at
# the centre of the screen) has user32 follow the mouse and send
# WM_VSCROLL.
features = test-smss
requires = userland, native
initial-command = SCROLL1.EXE
cpus     = default, intel
timeout  = 120
keys = [scroll] waiting => move:-76,-13 move:-76,-14 press:1 release

expect = [scroll] on WinSta0\Default
expect = [scroll] C, without scroll bars: GetScrollInfo 0 (1447), GetScrollPos 0 (1447)
expect = [scroll] A, from WM_NCCREATE: GetScrollRange 0 100, GetScrollPos 0
expect = [scroll] A: SetScrollInfo 30, GetScrollPos 30, GetScrollRange 0 99, GetScrollInfo 1: 0 99 10 30
expect = [scroll] A: SetScrollPos(200) 30, then 90
# The bar on the screen: right of the client area (104, 124, 480, 380), a
# pixel longer at each end (WS_BORDER); arrows of SM_CYVSCROLL; the thumb a
# page long (226 x 10 / 100, 23 pixels), at the end of the room.
expect = [scroll] A: GetScrollBarInfo 1: 480 123 496 381, arrows 16, thumb 219 242, state 0
# B's client area: 172 high without its bar, 156 with it.
expect = [scroll] B: client height 156; nothing to scroll: WS_HSCROLL 0, client height 172; again: WS_HSCROLL 1, client height 156
expect = [scroll] B, SIF_DISABLENOSCROLL: WS_HSCROLL 1, states 1 1 1; EnableScrollBar(ESB_ENABLE_BOTH) 1: 0 0; ESB_DISABLE_LTUP 1: 1 0
expect = [scroll] B: ShowScrollBar(FALSE) 1, client height 172, ShowScrollBar(TRUE) 1, client height 156, again 0
expect = [scroll] waiting
expect = [scroll] WM_VSCROLL SB_LINEDOWN: no control 1, the position before 30
expect = [scroll] WM_VSCROLL SB_ENDSCROLL: moved 1
expect = [scroll] A after the click: arrow states 0 0, thumb lower 1
expect = [scroll] done

forbid = *** STOP:

forbid = NtUserSBGetParms (0x11f0) is not implemented
forbid = NtUserGetScrollBarInfo (0x11af) is not implemented
forbid = NtUserSetScrollBarInfo (0x12a7) is not implemented
forbid = NtUserShowScrollBar (0x1226) is not implemented
forbid = NtUserEnableScrollBar (0x116e) is not implemented
forbid = win32k: painting: scroll bars
