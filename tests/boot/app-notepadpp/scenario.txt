# M7 bench: Notepad++ 7.9.2 (NSIS), installed silently by APPBENCH.EXE
# (tests/native/appbench.c) from C:\APPS, then started and closed; the
# bench runs as the shell of the logged-on user (session).
# The markers say how far it goes; docs/compat/applications.md has the
# report.
features = test-smss, interactive
requires = userland, native
session = APPBENCH.EXE
boots = 2
app = notepadpp
cpus = default
timeout = 900
# What works today, a floor: it installs, and in the user's session
# (session: lsass, the providers minoinit registers) its Authenticode
# check of its own DLLs passes; its main window opens (WM_CREATE asks
# SystemNumaProcessorMap through kernel32) and closes with 0. Its route
# then goes through its windows (route-step in xtask/src/apps.rs): it opens
# C:\APPS\ROUTE.TXT ("Mino route"), a line is typed at its end, Ctrl+H
# replaces "fox" with "cat" (Replace All), Ctrl+S saves; opened again, its
# text goes to the clipboard (Ctrl+A, Ctrl+C), which the bench reads.

keys = [appbench] route step: replace dialog => "fox" tab "cat" alt-a esc
keys = [appbench] route step: replaced => ctrl-s
keys = [appbench] route step: reopened => ctrl-a ctrl-c
