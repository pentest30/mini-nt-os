# through kernel32 (LockFile, LockFileEx, UnlockFile), the native services
# (status block, event, key), reads and writes stopped by a lock, the locks
# of a handle gone with it; then ole32's structured storage, which msiexec
# opens its package with (StgOpenStorage locks ranges past the end of the
# file and loops while they conflict). And CreateDirectoryW with a trailing
# backslash, which msi makes C:\WINDOWS\Installer\ with.

expect = [lock] LockFileEx: shared locks share, writes stop at them
expect = [lock] NtReadFile, NtWriteFile: an exclusive lock of another key stops them
expect = [lock] NtLockFile, NtUnlockFile: status block, event, key
expect = [lock] UnlockFile: the exact range
expect = [lock] the locks of a handle go with it
expect = [lock] ole32: StgCreateDocfile
