# The native installer UI, wired to the real engine: SETUP1.EXE takes the
# interactive desktop, renders the five-screen flow with custom GDI
# controls, and uses the keyboard path from main. A blank secondary ATA
# disk feeds the read-only kernel plan; activating Install sends the real
# IOCTL_MINO_SETUP_PREPARE with the token of the preview: the kernel
# revalidates the disk, writes the protective MBR and both GPT copies, and
# rereads them (the boot's own self-test stays out, behind the test-install
# feature, so the target is blank). The host then validates both copies and
# confirms the copied boot and system trees and every FAT32 cluster chain.
features = test-smss
requires = userland, native
initial-command = SETUP1.EXE
disk = target
cpus = default, intel
timeout = 900

expect = [setup] screen welcome
expect = [setup] backend target 2: 128 MiB; plan EFI 56 MiB, system 70 MiB; token
expect = [setup] focus back
expect = [setup] back to welcome
expect = [setup] screen disk 2
expect = [setup] returned to disk
expect = [setup] screen review 2
expect = [setup] screen install; GPT write boundary
expect = [setup] stage prepare: 20%
expect = [setup] prepare: GPT written and reread by the kernel
expect = [setup] stage format: 40%
expect = [setup] format: ESP and system volumes FAT32, mounted by the kernel
expect = [setup] stage copy: 60%
expect = [setup] copy: boot files and system tree copied
expect = [setup] stage configure: 80%
expect = [setup] stage verify: 100%
expect = [setup] screen complete; GPT, FAT32 and files written

# Final screen, in Mino's blue theme: the dark blue top band, the target-disk
# panel on the blue field, the pale-green completion card of the wizard.
screen = 900 30 062A87
screen = 280 410 2356CA
screen = 1100 320 ECF9EC

keys = [setup] ready => ret
keys = [setup] screen disk 1 => tab ret
keys = [setup] back to welcome => ret
keys = [setup] screen disk 2 => ret
keys = [setup] screen review 1 => esc
keys = [setup] returned to disk => ret
keys = [setup] screen review 2 => ret

# The engine refused nothing here: the target was blank and the token was
# the preview's own.
forbid = [setup] prepare failed:
forbid = [setup] format failed:
forbid = [setup] copy failed:
# The copy of the system tree lasts more than 10 s after the last key: the
# probe of test-smss must count the steps of the engine (io::install::
# last_step_time) as keys, not end SETUP1.EXE in the middle of the copy.
forbid = did not end within 10 s
