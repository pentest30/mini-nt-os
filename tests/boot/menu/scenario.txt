# Menus that user32 reads itself (win32k::menu: MENU and ITEM in the
# desktop's heap; tests/native/menu1.c, MinGW, linked to ReactOS's kernel32
# and user32): MENU1.EXE takes the place of winlogon.exe and switches to
# WinSta0\Default. It builds a bar (File, Edit, Help), gives it to W (100,
# 100, 400 x 300, white) and reads every change back through user32. Then
# TrackPopupMenuEx (win32k::menutrack, the window of the class #32768 in
# win32k::menupopup): the keys below, typed at each "[menu] tracking N" (the
# owner's first WM_ENTERIDLE), the mouse by messages W posts to the menu's
# window, EndMenu; the last menu stays open for the capture.
features = test-smss
requires = userland, native
initial-command = MENU1.EXE
cpus     = default, intel
timeout  = 120

expect = [menu] on WinSta0\Default
expect = [menu] CreateMenu: IsMenu 1 1, 0 items
expect = [menu] items: bar 3, File 3; GetSubMenu: File 1, Edit 1, Help none 1
# A popup item has no identifier for GetMenuItemID; a separator's is 0.
expect = [menu] GetMenuItemID: 101 0 -1
# MF_SEPARATOR | MF_GRAYED | MF_DISABLED; three items << 8 | MF_POPUP; MF_HELP.
expect = [menu] GetMenuState: separator 803, File popup 310, Exit by command 0, Help 4000
expect = [menu] GetMenuStringW: &File 5, by command E&xit 5
expect = [menu] GetMenuItemInfoW 1: submenu Edit 1, id 1, text &Edit 5, type 0, state 0
# Before the item 102, and at position 0.
expect = [menu] InsertMenuW: 100 101 0 103 102

# MFT_RADIOCHECK | MFS_CHECKED.

# (1400).

# The bar (win32k::menubar): SM_CYMENU high, between the frame's sides,
# over the client area; its items from its left, Help (MF_HELP) at its right.
expect = [menu] GetMenuBarInfo 1: menu 1, 19 high, the frame's width each side 1, the client area under it 1
expect = [menu] GetMenuItemRect 1 1 1: File at the bar's left 1, Edit after it 1, Help at its right 1, 18 high

# Read back from the screen: the text (more than 20 pixels that are not
# COLOR_MENU inside the item); hilited, a sunken edge (COLOR_BTNSHADOW at its
# corner) and the text still there.
expect = [menu] the text of File drawn 1; HiliteMenuItem: 80, sunken 1, text 1; unlit: 0, flat 1
expect = [menu] ModifyMenuW: Edit stays 1 until DrawMenuBar, then moves right 1
expect = [menu] RemoveMenu 1: 2 items, Edit lives 1; DeleteMenu 1: Edit gone 1
expect = [menu] LoadMenuIndirectW: 1 item, View 2 items, 402 &Small
expect = [menu] DestroyMenu 1: gone 1, its submenu gone 1
# ERROR_INVALID_FLAGS, ERROR_INVALID_MENU_HANDLE, ERROR_INVALID_PARAMETER.
expect = [menu] TrackPopupMenuEx refuses: a flag 0 (1004), a menu 0 (1401), a TPMPARAMS 0 (87)
# Shown, notified (WM_ENTERMENULOOP, WM_INITMENU, WM_INITMENUPOPUP,
# WM_UNINITMENUPOPUP, WM_EXITMENULOOP), but nothing to wait for.
expect = [menu] an empty menu: 0, notified 1 1 1 1 1, never waits 1
# Open, Gray (the separator skipped), Sub, its submenu on Inner, Enter.
keys = [menu] tracking 1 => down down down right ret
# The menu at (150, 160), its frame SM_CXDLGFRAME wide; shown already:
# ERROR_POPUP_ALREADY_ACTIVE.
expect = [menu] tracked 1: MSGF_MENU 1, MN_GETHMENU 1, at 150,160 1, a dialog's frame 1, again 0 (1446)
# WM_MENUSELECT: Open, Gray, Sub, Inner, then (0, 0xFFFF) and no menu.
expect = [menu] TrackPopupMenuEx 1: 301; ENTERMENULOOP 1 1, INITMENU 1 1, INITMENUPOPUP 2 1 1, MENUSELECT 5 ffff0000 0, UNINITMENUPOPUP 2, EXITMENULOOP 1, its window gone 1, WM_COMMAND 0
# Sub's submenu opened and closed twice (Escape, left: the tracking goes
# on); q: no item (WM_MENUCHAR 'q' | MF_POPUP << 16, ignored); x: Exit.
keys = [menu] tracking 2 => down down down right esc right left q x
expect = [menu] TrackPopupMenu 2: 1; INITMENUPOPUP 3, UNINITMENUPOPUP 3, WM_MENUCHAR 100071, WM_COMMAND 203
# W posts the mouse's messages to the menu's window, at the screen's
# coordinates (the menu tracked: not the client area's).
expect = [menu] TrackPopupMenuEx 3, the mouse: 201; Open hilited 1, its text 1, the separator 1; Sub opens 1, at the right 1, WM_ENTERIDLE names it 1
# To the right of the rectangle to avoid (100, 150, 300, 250).
expect = [menu] TrackPopupMenuEx 4, EndMenu: 0; at 300,160; notified 0
# The mouse, from the middle of the screen (640, 400), to Sub's (201, 214):
# its submenu opens (the menu tracked, the mouse's messages keep the screen's
# coordinates).
keys = [menu] tracking 5 => move:-220,-93 move:-219,-93
# The places the screen's checks below rely on: the menu, Sub (in its client
# area, 3 pixels in), its submenu's Inner to the right of the menu.
expect = [menu] the last menu at 150,160 to 253,243; Sub 153,206 to 250,223; Inner 252,206 to 309,223
expect = [menu] done
