# The scrolling of a window and of a DC (win32k::scrollex; tests/native/
# scrollx1.c, MinGW, linked to ReactOS's kernel32, user32 and gdi32):
# SCROLLX1.EXE takes the place of winlogon.exe and switches to
# WinSta0\Default. W (100, 100, 400 x 300; its client area 392 x 272 from
# 104, 124) shows bands of 20 pixels, red, green, blue, from a position
# `top`, and paints only what it is asked to. ScrollWindowEx up 30 uncovers
# the last 30 rows (SIMPLEREGION), which alone are painted; ScrollDC down 20
# on its DC uncovers the first 20; with SW_SCROLLCHILDREN left 10, its
# child C (50 x 30 at 300, 100) moves to 290, 100 and receives WM_MOVE.
features = test-smss
requires = userland, native
initial-command = SCROLLX1.EXE
cpus     = default, intel
timeout  = 120

expect = [scrollx] on WinSta0\Default
expect = [scrollx] painted
expect = [scrollx] ScrollWindowEx up 30: 2, prcUpdate 0 242 392 272, hrgnUpdate 0 242 392 272, GetUpdateRect 0 242 392 272
expect = [scrollx] WM_PAINT 0 242 392 272
expect = [scrollx] ScrollDC down 20: 1, prcUpdate 0 0 392 20
expect = [scrollx] WM_PAINT 0 0 392 20
expect = [scrollx] SW_SCROLLCHILDREN left 10: 2, C from 404 to 394, WM_MOVE 290 100
expect = [scrollx] done

forbid = *** STOP:
forbid = NtUserScrollWindowEx (0x11f2) is not implemented
forbid = NtUserScrollDC (0x11f1) is not implemented
